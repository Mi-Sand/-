# Зависимости проекта.
#
# Версии заданы диапазонами, а не точными номерами. Точная привязка
# однажды уже подвела: django-filter 23.2 вызывал pkgutil.find_loader,
# который в Python 3.14 удалён, и проект переставал запускаться вовсе.
# Диапазон позволяет поставить исправленную версию, но не пускает
# следующее крупное обновление, способное сломать совместимость.
#
# Проверено: 191 тест проходит на Python 3.14 с Django 5.2 LTS и с 6.0.

# Django 5.2 LTS или 6.0.
#
# Проект переведён с 4.2 намеренно. Во-первых, поддержка 4.2 LTS
# закончилась в апреле 2026 года — обновления безопасности для неё больше
# не выходят. Во-вторых, Python 3.14 официально поддерживается только
# начиная с Django 6.0, а на 4.2 работал случайно.
#
# Обратной совместимости с 4.2 больше нет: в ограничениях моделей
# используется CheckConstraint(condition=...), появившийся в Django 5.1
# взамен прежнего check=. На 4.2 проект теперь не запустится.
Django>=5.2,<7

# 3.17 — первая версия с поддержкой Django 6.
djangorestframework>=3.17,<4

# До 23.5 использовался pkgutil.find_loader, удалённый в Python 3.14.
# Версия 25.1 — последняя, работающая с Django 4.2; начиная с 26.0
# требуется Django 5.1+, что нам подходит.
django-filter>=25.1,<27

django-cors-headers>=4.0,<5

# Драйвер PostgreSQL. Используется psycopg 3 (не psycopg2):
# он ставится готовым бинарником и не требует компиляции под Windows.
psycopg[binary]>=3.1

python-dotenv>=1.0,<2

# Экспорт отчётов и документов бухгалтерии
openpyxl>=3.1,<4
reportlab>=4.0,<6

# Обработка изображений — обязательна для ImageField (фото товаров)
Pillow>=10.0

# WSGI-сервер для боевого режима. Именно его запускает Dockerfile и
# docker-compose — без него контейнер не стартует.
gunicorn>=21.2
