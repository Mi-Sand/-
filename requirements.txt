# Зависимости проекта.
#
# Версии заданы диапазонами, а не точными номерами. Точная привязка
# однажды уже подвела: django-filter 23.2 вызывал pkgutil.find_loader,
# который в Python 3.14 удалён, и проект переставал запускаться вовсе —
# с ошибкой, по которой причина не видна. Диапазон позволяет поставить
# исправленную версию, но не пускает следующее крупное обновление,
# способное сломать совместимость.
#
# Проверено на Python 3.11 и 3.14: 191 тест проходит на обеих.

# Django 4.2 LTS. Начиная с 4.2.27 исправлена совместимость с Python
# 3.13 и 3.14.
Django>=4.2.27,<4.3

# 3.15 — первая версия с поддержкой современных Python.
djangorestframework>=3.15,<4

django-cors-headers>=4.0,<5

# Нижняя граница важна: до 23.5 использовался pkgutil.find_loader,
# удалённый в Python 3.14. Верхняя — тоже: начиная с 26.0 требуется
# Django 5, несовместимая с нашей 4.2.
django-filter>=23.5,<26

# Драйвер PostgreSQL. Используется psycopg 3 (не psycopg2):
# он ставится готовым бинарником и не требует компиляции под Windows.
psycopg[binary]>=3.1

python-dotenv>=1.0,<2

# Экспорт отчётов и документов бухгалтерии
openpyxl>=3.1,<4
reportlab>=4.0,<6

# Обработка изображений — обязательна для ImageField (фото товаров)
Pillow>=10.0

# WSGI-сервер для боевого режима. Именно его запускает Dockerfile и
# docker-compose — без него контейнер не стартует.
gunicorn>=21.2
