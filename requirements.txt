# Зависимости проекта.
#
# Версии заданы диапазонами, а не точными номерами. Точная привязка
# однажды уже подвела: django-filter 23.2 вызывал pkgutil.find_loader,
# удалённый в Python 3.14, и проект переставал запускаться вовсе.
# Диапазон позволяет поставить исправленную версию, но не пускает
# следующее крупное обновление, способное сломать совместимость.

# Django 5.2 LTS или 6.0.
#
# Поддержка 4.2 LTS закончилась в апреле 2026 года, а Python 3.14
# официально поддерживается только начиная с Django 6.0. В ограничениях
# моделей используется CheckConstraint(condition=...), появившийся в
# Django 5.1 взамен прежнего check=, — на 4.2 проект не запустится.
#
# Верхняя граница — не перестраховка: в Django 6.1 из django.utils.cache
# убрали cc_delim_re, который импортирует DRF, и приложение падает при
# запуске. Снимать её можно будет, когда выйдет поддерживающая 6.1 версия
# DRF (на момент проверки новейшая — 3.17.2, она с 6.1 не работает).
Django>=5.2,<6.1

djangorestframework>=3.17,<4

# До 23.5 использовался pkgutil.find_loader, удалённый в Python 3.14.
django-filter>=25.1,<27

django-cors-headers>=4.0,<5

# Драйвер PostgreSQL. Используется psycopg 3 (не psycopg2):
# он ставится готовым бинарником и не требует компиляции под Windows.
psycopg[binary]>=3.1

python-dotenv>=1.0,<2

# Экспорт отчётов
openpyxl>=3.1,<4
reportlab>=4.0,<6

# Обработка изображений — обязательна для ImageField (фото товаров)
Pillow>=10.0

# WSGI-сервер для боевого режима. Именно его запускает Dockerfile —
# без него контейнер не стартует.
gunicorn>=21.2

# Раздача статики самим приложением. Нужна, когда перед Django нет nginx:
# при запуске на офисном компьютере интерфейс иначе останется без стилей.
whitenoise>=6.6
